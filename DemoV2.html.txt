<script>
// Global audio player
let currentAudio = new Audio();

function playSound(fileName) {
  if (currentAudio) {
    currentAudio.pause();          // Immediately cuts off any active playback
    currentAudio.currentTime = 0;  // Rewinds track to the beginning
  }
  currentAudio.src = fileName;
  currentAudio.play();
}  // Simple Browser Speech Synthesis (Reads text aloud to mimic IVR prompts)
  function speak(text) {
    if ('speechSynthesis' in window) {
      window.speechSynthesis.cancel();
      const utterance = new SpeechSynthesisUtterance(text);
      utterance.rate = 1.0;
      window.speechSynthesis.speak(utterance);
    }
  }

  function setStatus(msg) {
    document.getElementById('statusBox').innerText = msg;
  }

function startCall() {
  document.getElementById('callBtn').classList.add('hidden');
  document.getElementById('menuScreen').classList.remove('hidden');
  document.getElementById('routedScreen').classList.remove('hidden');
  
  setStatus("Call Connected. Listening to automated menu...");
  
  // Play your actual greeting recording:
  playSound('greeting.mp3');
}

 function handleMenuSelection(optionNum, department, audioFile) {
  // 1. Cut off the greeting immediately
  currentAudio.pause();
  currentAudio.currentTime = 0;
  if ('speechSynthesis' in window) window.speechSynthesis.cancel();

  const response = `You selected option ${optionNum} for ${department}. Connecting you to a representative now.`;
  setStatus(response);

  // 2. Play the new audio clip if provided, otherwise fallback to speak
  if (audioFile) {
    playSound(audioFile);
  } else {
    speak(response);
  }
}

  function showVoiceInputScreen() {
    document.getElementById('menuScreen').classList.add('hidden');
    document.getElementById('voiceScreen').classList.remove('hidden');
    
    const promptMsg = "Please say or type the tax type you want to talk to a specialist about.";
    setStatus(promptMsg);
    speak(promptMsg);
  }

  function handleRoutingSuccess(taxType) {
    document.getElementById('voiceScreen').classList.add('hidden');
    const transferMsg = `Routing your call to a specialist in ${taxType}. Please hold.`;
    setStatus(transferMsg);
    speak(transferMsg);
  }

  function submitManualInput() {
    const val = document.getElementById('manualInput').value.trim();
    if (val) {
      handleRoutingSuccess(val);
    }
  }

  // Voice Recognition Handling
  let recognition;
  let isListening = false;

  if ('webkitSpeechRecognition' in window || 'SpeechRecognition' in window) {
    const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
    recognition = new SpeechRecognition();
    recognition.continuous = false;
    recognition.interimResults = false;
    recognition.lang = 'en-US';

    recognition.onstart = function() {
      isListening = true;
      const mic = document.getElementById('micBtn');
      mic.classList.add('listening');
      mic.innerText = "Listening... Speak now";
    };

    recognition.onresult = function(event) {
      const transcript = event.results[0][0].transcript;
      handleRoutingSuccess(transcript);
    };

    recognition.onerror = function(event) {
      console.error(event.error);
      setStatus("Could not hear clearly. Please try speaking again or use the text box.");
      resetMicUI();
    };

    recognition.onend = function() {
      resetMicUI();
    };
  }

  function resetMicUI() {
    isListening = false;
    const mic = document.getElementById('micBtn');
    mic.classList.remove('listening');
    mic.innerText = "🎤 Tap to Speak Tax Type";
  }

  function toggleVoiceRecognition() {
    if (!recognition) {
      alert("Voice recognition is not supported on this browser version. Use the text box below.");
      return;
    }
    if (isListening) {
      recognition.stop();
    } else {
      recognition.start();
    }
  }

  function resetDemo() {
    if ('speechSynthesis' in window) window.speechSynthesis.cancel();
    if (recognition && isListening) recognition.stop();

    document.getElementById('callBtn').classList.remove('hidden');
    document.getElementById('menuScreen').classList.add('hidden');
    document.getElementById('voiceScreen').classList.add('hidden');
    document.getElementById('routedScreen').classList.add('hidden');
    document.getElementById('manualInput').value = '';
    setStatus("Press \"Call DOR\" to start the simulated call.");
  }
</script>
